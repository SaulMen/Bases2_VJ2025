db.aspirantes.aggregate([
  {
    $addFields: {
      edad: { $subtract: [2025, "$anio_nacimiento"] },
    },
  },
  {
    $group: {
      _id: "$edad",
      total_evaluaciones: { $sum: 1 },
      total_aprobados: { $sum: { $cond: ["$aprobacion", 1, 0] } },
    },
  },
  {
    $project: {
      _id: 0,
      edad: "$_id",
      total_evaluaciones: 1,
      total_aprobados: 1,
      tasa_aprobacion: {
        $cond: {
          if: { $eq: ["$total_evaluaciones", 0] },
          then: 0, 
          else: { $multiply: [{ $divide: ["$total_aprobados", "$total_evaluaciones"] }, 100] }
        }
      },
    },
  },
  {
    $sort: {
      edad: 1,
    },
  },
]);
